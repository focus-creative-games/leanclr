cmake_minimum_required(VERSION 3.15)
project(aot_return_protocol_tests CXX)
add_executable(aot-returns main.cpp)
target_include_directories(aot-returns PRIVATE ../../runtime)
set_target_properties(aot-returns PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
enable_testing()
if(EMSCRIPTEN)
    set_target_properties(aot-returns PROPERTIES SUFFIX ".js")
    target_link_options(aot-returns PRIVATE -sENVIRONMENT=node -sFILESYSTEM=0)
endif()
add_test(NAME aot-returns COMMAND aot-returns)
