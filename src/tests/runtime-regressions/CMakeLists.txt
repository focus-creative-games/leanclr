cmake_minimum_required(VERSION 3.15)
project(runtime_regressions C CXX)
add_subdirectory(../../runtime runtime_build)
add_executable(runtime-regressions main.cpp)

set_target_properties(runtime-regressions PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
set(LEANCLR_TEST_DLL_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../out/cmake/tests/basic-tester/Debug-x64/bin/Debug/dlls" CACHE PATH "Managed test and BCL directory")
enable_testing()
add_test(NAME zero-arrays COMMAND runtime-regressions zero-arrays "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME clone-arrays COMMAND runtime-regressions clone-arrays "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME metadata-bounds COMMAND runtime-regressions metadata-bounds "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME exported-types COMMAND runtime-regressions exported-types "${LEANCLR_TEST_DLL_DIR}")
