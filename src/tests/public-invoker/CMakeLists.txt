cmake_minimum_required(VERSION 3.15)
project(public_invoker_contract C CXX)
if(NOT TARGET leanclr)
    add_subdirectory(../../runtime runtime_build)
endif()
add_executable(public-invoker native.cpp callbacks.c)
target_link_libraries(public-invoker PRIVATE leanclr)
set_target_properties(public-invoker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(EMSCRIPTEN)
    set_target_properties(public-invoker PROPERTIES SUFFIX ".js")
    target_link_options(public-invoker PRIVATE -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
endif()
set(LEANCLR_PUBLIC_INVOKER_RUNTIME_DIR "" CACHE STRING "Matching BCL directory")
enable_testing()
add_test(NAME public-invoker COMMAND public-invoker "${LEANCLR_PUBLIC_INVOKER_RUNTIME_DIR}")
set_tests_properties(public-invoker PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
