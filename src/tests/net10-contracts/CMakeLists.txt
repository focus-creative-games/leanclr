cmake_minimum_required(VERSION 3.15)
project(net10_contracts C CXX)
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "BCL adapter")
if(NOT LEANCLR_BCL_ADAPTER STREQUAL "CORECLR_NET10")
    message(FATAL_ERROR "These contracts require CORECLR_NET10")
endif()
add_subdirectory(../../runtime runtime_build)
enable_testing()
add_executable(net10-empty-ranges ../runtime-regressions/container-allocation.cpp)
target_link_libraries(net10-empty-ranges PRIVATE leanclr)
add_test(NAME net10-empty-ranges COMMAND net10-empty-ranges)
add_executable(net10-handles dependent-handles.cpp)
target_link_libraries(net10-handles PRIVATE leanclr)
add_test(NAME net10-handles COMMAND net10-handles)

add_executable(net10-native-handles native-handles.cpp)
target_link_libraries(net10-native-handles PRIVATE leanclr)
add_test(NAME net10-native-handles COMMAND net10-native-handles)

add_executable(net10-resources resources.cpp)
target_link_libraries(net10-resources PRIVATE leanclr)
if(WIN32)
    target_link_libraries(net10-resources PRIVATE psapi)
endif()
set(LEANCLR_NET10_RUNTIME_DIR "" CACHE STRING "Optional CoreLib directory for real-heap resource checks")
add_test(NAME net10-resources COMMAND net10-resources 1024 1024 "${LEANCLR_NET10_RUNTIME_DIR}")
set_tests_properties(net10-resources PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
if(EMSCRIPTEN)
    set_target_properties(net10-empty-ranges net10-handles net10-native-handles net10-resources PROPERTIES SUFFIX ".js")
    target_link_options(net10-resources PRIVATE -O2 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
endif()

add_executable(net10-environment environment.cpp)
target_link_libraries(net10-environment PRIVATE leanclr)
add_test(NAME net10-environment COMMAND net10-environment)
if(EMSCRIPTEN)
    set_target_properties(net10-environment PROPERTIES SUFFIX ".js")
    target_link_options(net10-environment PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
endif()
