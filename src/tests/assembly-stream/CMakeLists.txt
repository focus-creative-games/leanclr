cmake_minimum_required(VERSION 3.15)
project(assembly_stream_contracts C CXX)
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "BCL adapter")
add_subdirectory(../../tools/leanrun leanrun_build)
add_executable(assembly-stream-memory native.cpp)
target_link_libraries(assembly-stream-memory PRIVATE leanclr)
set(LEANCLR_STREAM_MANAGED_DIR "" CACHE PATH "Managed test assembly directory")
set(LEANCLR_STREAM_RUNTIME_DIR "" CACHE PATH "Matching CoreLib directory")
if(EMSCRIPTEN)
    foreach(target leanrun assembly-stream-memory)
        set_target_properties(${target} PROPERTIES SUFFIX ".js")
        target_link_options(${target} PRIVATE -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=1 -lnodefs.js
            "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
    endforeach()
endif()
enable_testing()
foreach(entry DefaultOnly Memory Exposed Offset Duplicate Conflict Symbols BadSymbols Forward Invalid Empty Closed Null Lifetime BeforeDependency EquivalentImage ShortReads ReadFailure UnsupportedContexts MissingDependency Truncated)
    add_test(NAME stream-${entry} COMMAND leanrun -l "${LEANCLR_STREAM_MANAGED_DIR}" -l "${LEANCLR_STREAM_RUNTIME_DIR}"
        -e "AssemblyStreamTests.Program::${entry}" AssemblyStreamTests)
    set_tests_properties(stream-${entry} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
endforeach()
if(WIN32 AND NOT LEANCLR_FORCE_PORTABLE AND LEANCLR_HOST STREQUAL "NATIVE")
    foreach(entry FileStream InvalidFileStream)
        add_test(NAME stream-${entry} COMMAND leanrun -l "${LEANCLR_STREAM_MANAGED_DIR}" -l "${LEANCLR_STREAM_RUNTIME_DIR}"
            -e "AssemblyStreamTests.Program::${entry}" AssemblyStreamTests)
        set_tests_properties(stream-${entry} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
    endforeach()
endif()
add_test(NAME stream-native-memory COMMAND assembly-stream-memory "${LEANCLR_STREAM_MANAGED_DIR}" "${LEANCLR_STREAM_RUNTIME_DIR}")
set_tests_properties(stream-native-memory PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
