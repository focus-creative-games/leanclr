cmake_minimum_required(VERSION 3.15)
project(eh_aot_tests C CXX)
set(EH_AOT_CPP "" CACHE PATH "Generated fixture sources")
set(EH_AOT_DLLS "" CACHE STRING "Fixture and BCL directory (host/VFS path)")
add_subdirectory(../../runtime runtime)
file(GLOB generated CONFIGURE_DEPENDS "${EH_AOT_CPP}/*.cpp")
add_executable(eh-aot main.cpp ${generated})
target_include_directories(eh-aot PRIVATE "${EH_AOT_CPP}")
target_link_libraries(eh-aot PRIVATE leanclr)
set_target_properties(eh-aot PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
enable_testing()
if(EMSCRIPTEN)
    set_target_properties(eh-aot PROPERTIES SUFFIX ".js")
    target_link_options(eh-aot PRIVATE -O2 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
endif()
add_test(NAME eh-aot COMMAND eh-aot "${EH_AOT_DLLS}")
set_tests_properties(eh-aot PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
