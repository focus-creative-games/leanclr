cmake_minimum_required(VERSION 3.15)
project(leanclr_result_abi_bench CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BENCH_LTO "Enable interprocedural optimization" OFF)
set(BENCH_OPT "2" CACHE STRING "Optimization level: 2, 3 or s")
if(NOT BENCH_OPT MATCHES "^(2|3|s)$")
    message(FATAL_ERROR "BENCH_OPT must be 2, 3 or s")
endif()
if(BENCH_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT ipo_ok OUTPUT ipo_error)
    if(NOT ipo_ok)
        message(FATAL_ERROR "IPO unavailable: ${ipo_error}")
    endif()
endif()
enable_testing()
set(protocols current trivial scalar status context eh direct)
set(protocol_id 0)
foreach(protocol ${protocols})
    add_executable(result-${protocol} main.cpp kernels.cpp)
    target_include_directories(result-${protocol} PRIVATE ../../runtime)
    target_compile_definitions(result-${protocol} PRIVATE NDEBUG BENCH_PROTOCOL=${protocol_id} BENCH_NAME="${protocol}")
    if(BENCH_LTO)
        set_property(TARGET result-${protocol} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
    if(MSVC)
        if(BENCH_OPT STREQUAL "s")
            target_compile_options(result-${protocol} PRIVATE /O1)
        else()
            target_compile_options(result-${protocol} PRIVATE /O2)
        endif()
        target_compile_options(result-${protocol} PRIVATE /EHsc)
    else()
        target_compile_options(result-${protocol} PRIVATE -O${BENCH_OPT} -fexceptions)
        target_link_options(result-${protocol} PRIVATE -O${BENCH_OPT})
    endif()
    if(EMSCRIPTEN)
        set_target_properties(result-${protocol} PROPERTIES SUFFIX ".js")
        target_compile_options(result-${protocol} PRIVATE -fwasm-exceptions)
        target_link_options(result-${protocol} PRIVATE -fwasm-exceptions -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=0 -sFILESYSTEM=0)
    endif()
    add_test(NAME verify-${protocol} COMMAND result-${protocol} --verify)
    math(EXPR protocol_id "${protocol_id} + 1")
endforeach()
