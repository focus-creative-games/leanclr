cmake_minimum_required(VERSION 3.15)
project(platform_contracts C CXX)
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "BCL adapter")
add_subdirectory(../../runtime runtime_build)
set(PLATFORM_AOT_CPP "" CACHE PATH "Generated contract sources")
file(GLOB generated CONFIGURE_DEPENDS "${PLATFORM_AOT_CPP}/*.cpp")
add_executable(platform-interp ../../tools/leanrun/main.cpp)
add_executable(platform-aot ../aot-tester/main.cpp ../aot-tester/pinvoke_impl.cpp ${generated})
target_compile_definitions(platform-aot PRIVATE LEANCLR_REQUIRE_AOT_ENTRY=1)
add_executable(platform-environment ../net10-contracts/environment.cpp)
foreach(target platform-interp platform-aot platform-environment)
    target_link_libraries(${target} PRIVATE leanclr)
    if(EMSCRIPTEN)
        set_target_properties(${target} PROPERTIES SUFFIX ".js")
        target_link_options(${target} PRIVATE -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=1)
        if(NOT target STREQUAL "platform-environment")
            target_link_options(${target} PRIVATE -lnodefs.js
                "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
        endif()
    endif()
endforeach()
enable_testing()
add_test(NAME platform-environment COMMAND platform-environment)

add_subdirectory(../public-invoker public-invoker)
