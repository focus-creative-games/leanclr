cmake_minimum_required(VERSION 3.15)
project(runtime_regressions C CXX)
add_subdirectory(../../runtime runtime_build)
enable_testing()
add_executable(public-c public-c.c)
target_link_libraries(public-c PRIVATE leanclr)
set_target_properties(public-c PROPERTIES LINKER_LANGUAGE CXX)
add_executable(public-cpp public-cpp.cpp)
target_link_libraries(public-cpp PRIVATE leanclr)
add_test(NAME public-c COMMAND public-c)
add_test(NAME public-cpp COMMAND public-cpp)
if(BUILD_SHARED_LEANCLR AND WIN32)
    foreach(consumer public-c public-cpp)
        add_custom_command(TARGET ${consumer} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:leanclr>" "$<TARGET_FILE_DIR:${consumer}>")
    endforeach()
endif()
option(LEANCLR_REGRESSION_PUBLIC_API_ONLY "Only build standalone public ABI consumers" OFF)
if(LEANCLR_REGRESSION_PUBLIC_API_ONLY)
    return()
endif()
add_executable(integer-math integer-math.cpp)
target_include_directories(integer-math PRIVATE ../../runtime)
set_target_properties(integer-math PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
add_test(NAME integer-math COMMAND integer-math)
add_executable(runtime-regressions main.cpp)
target_link_libraries(runtime-regressions PRIVATE leanclr)
target_compile_definitions(runtime-regressions PRIVATE LEANCLR_REGRESSION_FIXTURE_DIR="${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(runtime-regressions PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
set(LEANCLR_REGRESSION_AOT_DIR "" CACHE PATH "Optional generated RegressionFixtures AOT sources")
if(LEANCLR_REGRESSION_AOT_DIR)
    file(GLOB fixture_aot_sources CONFIGURE_DEPENDS "${LEANCLR_REGRESSION_AOT_DIR}/*.cpp")
    target_sources(runtime-regressions PRIVATE ${fixture_aot_sources})
    target_include_directories(runtime-regressions PRIVATE "${LEANCLR_REGRESSION_AOT_DIR}")
    target_compile_definitions(runtime-regressions PRIVATE LEANCLR_REGRESSION_AOT=1)
endif()
add_executable(runtime-managed-tests ../basic-tester/main.cpp)
target_link_libraries(runtime-managed-tests PRIVATE leanclr)
set_target_properties(runtime-managed-tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
set(LEANCLR_TEST_DLL_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../out/cmake/tests/basic-tester/Debug-x64/bin/Debug/dlls" CACHE PATH "Managed test and BCL directory")
enable_testing()
find_program(LEANCLR_CSHARP_COMPILER NAMES csc mcs HINTS "C:/Windows/Microsoft.NET/Framework64/v4.0.30319")
if(LEANCLR_CSHARP_COMPILER)
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/RegressionFixtures.cs" fixture_source)
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_BINARY_DIR}/RegressionFixtures.dll" fixture_output)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/RegressionFixtures.dll"
        COMMAND "${LEANCLR_CSHARP_COMPILER}" -nologo -target:library "-out:${fixture_output}" "${fixture_source}"
        DEPENDS RegressionFixtures.cs VERBATIM)
    add_custom_target(regression-fixtures ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/RegressionFixtures.dll")
    add_dependencies(runtime-regressions regression-fixtures)
    add_test(NAME failed-initializer COMMAND runtime-regressions failed-initializer "${LEANCLR_TEST_DLL_DIR}")
    add_test(NAME invocation-buffers COMMAND runtime-regressions invocation-buffers "${LEANCLR_TEST_DLL_DIR}")
    add_test(NAME weak-finalizers COMMAND runtime-regressions weak-finalizers "${LEANCLR_TEST_DLL_DIR}")
    add_test(NAME allocation-failures COMMAND runtime-regressions allocation-failures "${LEANCLR_TEST_DLL_DIR}")
    add_test(NAME loader-once COMMAND runtime-regressions loader-once "${LEANCLR_TEST_DLL_DIR}")
    add_test(NAME integer-arithmetic COMMAND runtime-regressions integer-arithmetic "${LEANCLR_TEST_DLL_DIR}")
endif()
find_program(LEANCLR_ILASM NAMES ilasm HINTS "C:/Windows/Microsoft.NET/Framework64/v4.0.30319")
if(LEANCLR_ILASM)
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/FailedModule.il" module_source)
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_BINARY_DIR}/FailedModule.dll" module_output)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/FailedModule.dll"
        COMMAND "${LEANCLR_ILASM}" /dll "/output:${module_output}" "${module_source}"
        DEPENDS FailedModule.il VERBATIM)
    add_custom_target(regression-module ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/FailedModule.dll")
    add_dependencies(runtime-regressions regression-module)
    add_test(NAME failed-module COMMAND runtime-regressions failed-module "${LEANCLR_TEST_DLL_DIR}")
endif()
add_test(NAME zero-arrays COMMAND runtime-regressions zero-arrays "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME clone-arrays COMMAND runtime-regressions clone-arrays "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME metadata-bounds COMMAND runtime-regressions metadata-bounds "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME exported-types COMMAND runtime-regressions exported-types "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME gc-accounting COMMAND runtime-regressions gc-accounting "${LEANCLR_TEST_DLL_DIR}")
add_test(NAME compressed-signed COMMAND runtime-regressions compressed-signed "${LEANCLR_TEST_DLL_DIR}")
