cmake_minimum_required(VERSION 3.15)
project(leanclr_il_families C CXX)
set(LEANCLR_BENCH_GENERATED_DIR "" CACHE PATH "Generated AOT sources")
set(LEANCLR_GC_DEBUG OFF CACHE BOOL "Disable GC debugging for measurement")
add_subdirectory(../../runtime runtime)
file(GLOB generated CONFIGURE_DEPENDS "${LEANCLR_BENCH_GENERATED_DIR}/*.cpp")
add_executable(il-aot main.cpp ${generated})
target_compile_definitions(il-aot PRIVATE BENCH_AOT=1)
target_include_directories(il-aot PRIVATE "${LEANCLR_BENCH_GENERATED_DIR}")
target_link_libraries(il-aot PRIVATE leanclr)
add_executable(il-interp main.cpp)
target_link_libraries(il-interp PRIVATE leanclr)
