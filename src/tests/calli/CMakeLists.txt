cmake_minimum_required(VERSION 3.15)
project(calli_contracts C CXX)
add_subdirectory(../../tools/leanrun leanrun_build)
add_executable(calli-native native.cpp)
add_library(calli-fixture STATIC native_fixture.c)
target_link_libraries(calli-native PRIVATE leanclr calli-fixture)
if(NOT EMSCRIPTEN)
    add_library(calli-fixture-shared SHARED native_fixture.c)
endif()
set(LEANCLR_CALLI_BRIDGE_CPP_DIR "" CACHE PATH "Bridge-only generated sources")
if(LEANCLR_CALLI_BRIDGE_CPP_DIR AND NOT LEANCLR_CALLI_AOT_CPP_DIR)
    target_sources(calli-native PRIVATE "${LEANCLR_CALLI_BRIDGE_CPP_DIR}/calli_bridges.cpp")
    target_compile_definitions(calli-native PRIVATE LEANCLR_CALLI_BRIDGES=1)
endif()
set(LEANCLR_CALLI_AOT_CPP_DIR "" CACHE PATH "Optional generated calli sources")
if(LEANCLR_CALLI_AOT_CPP_DIR)
    file(GLOB calli_aot_sources CONFIGURE_DEPENDS "${LEANCLR_CALLI_AOT_CPP_DIR}/*.cpp")
    target_sources(calli-native PRIVATE ${calli_aot_sources})
    target_compile_definitions(calli-native PRIVATE LEANCLR_CALLI_AOT=1)
endif()
set(LEANCLR_CALLI_MANAGED_DIR "" CACHE PATH "Managed test assembly directory")
set(LEANCLR_CALLI_RUNTIME_DIR "" CACHE PATH "Matching BCL directory")
if(EMSCRIPTEN)
    foreach(target leanrun calli-native)
    set_target_properties(${target} PROPERTIES SUFFIX ".js")
    target_link_options(${target} PRIVATE -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
    endforeach()
endif()
enable_testing()
foreach(entry Managed ManagedRef ManagedGeneric FromHandle GenericCaller StructReturn RefReturn Mixed Collect StackRoot Exception BadTarget NullTarget WrongSignature)
    add_test(NAME calli-${entry} COMMAND leanrun -l "${LEANCLR_CALLI_MANAGED_DIR}" -l "${LEANCLR_CALLI_RUNTIME_DIR}"
        -e "CalliTests.Program::${entry}" CalliTests)
    set_tests_properties(calli-${entry} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
endforeach()

add_test(NAME calli-native COMMAND calli-native "${LEANCLR_CALLI_MANAGED_DIR}" "${LEANCLR_CALLI_RUNTIME_DIR}")
set_tests_properties(calli-native PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
