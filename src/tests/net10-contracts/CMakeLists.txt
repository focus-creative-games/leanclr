cmake_minimum_required(VERSION 3.15)
project(net10_contracts C CXX)
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "BCL adapter")
if(NOT LEANCLR_BCL_ADAPTER STREQUAL "CORECLR_NET10")
    message(FATAL_ERROR "These contracts require CORECLR_NET10")
endif()
add_subdirectory(../../runtime runtime_build)
enable_testing()
add_executable(net10-handles dependent-handles.cpp)
target_link_libraries(net10-handles PRIVATE leanclr)
add_test(NAME net10-handles COMMAND net10-handles)
