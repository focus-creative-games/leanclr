cmake_minimum_required(VERSION 3.15)
project(generic_aot_tests C CXX)
set(GENERIC_AOT_CPP "" CACHE PATH "Generated fixture sources")
set(GENERIC_AOT_DLLS "" CACHE STRING "Fixture and BCL directory (host/VFS path)")
option(GENERIC_AOT_DISABLED "Expect closed methods to use interpreter" OFF)
add_subdirectory(../../runtime runtime)
file(GLOB generated CONFIGURE_DEPENDS "${GENERIC_AOT_CPP}/*.cpp")
add_executable(generic-aot main.cpp ${generated})
target_include_directories(generic-aot PRIVATE "${GENERIC_AOT_CPP}")
target_link_libraries(generic-aot PRIVATE leanclr)
target_compile_features(generic-aot PRIVATE cxx_std_17)
enable_testing()
add_subdirectory(../aot-returns aot-returns)
if(EMSCRIPTEN)
    set_target_properties(generic-aot PROPERTIES SUFFIX ".js")
    target_link_options(generic-aot PRIVATE -O2 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
endif()
add_test(NAME generic-aot COMMAND generic-aot "${GENERIC_AOT_DLLS}" $<IF:$<BOOL:${GENERIC_AOT_DISABLED}>,interpreted,aot>)
set_tests_properties(generic-aot PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
