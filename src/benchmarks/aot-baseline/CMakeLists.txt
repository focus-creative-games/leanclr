cmake_minimum_required(VERSION 3.15)
project(leanclr_aot_baseline C CXX)
add_executable(cpp-reference reference.cpp)
set_target_properties(cpp-reference PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
set(LEANCLR_BENCH_GENERATED_DIR "" CACHE PATH "LeanAOT generated source directory")
if(NOT EXISTS "${LEANCLR_BENCH_GENERATED_DIR}")
    message(FATAL_ERROR "Set LEANCLR_BENCH_GENERATED_DIR")
endif()
add_subdirectory(../../runtime runtime)
file(GLOB generated CONFIGURE_DEPENDS "${LEANCLR_BENCH_GENERATED_DIR}/*.cpp")
add_executable(aot-baseline main.cpp ${generated})
target_include_directories(aot-baseline PRIVATE "${LEANCLR_BENCH_GENERATED_DIR}")
target_link_libraries(aot-baseline PRIVATE leanclr)
set_target_properties(aot-baseline PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
    set_target_properties(cpp-reference PROPERTIES SUFFIX ".js")
    target_compile_options(cpp-reference PRIVATE -O2)
    target_link_options(cpp-reference PRIVATE -O2 -sENVIRONMENT=node -sFILESYSTEM=0)
    set_target_properties(aot-baseline PROPERTIES SUFFIX ".js")
    target_link_options(aot-baseline PRIVATE -O2 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/nodefs.js")
endif()
