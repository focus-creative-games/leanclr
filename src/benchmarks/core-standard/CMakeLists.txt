cmake_minimum_required(VERSION 3.15)
project(core_standard_probe C CXX)
add_subdirectory(../../runtime runtime)
add_executable(core-standard main.cpp)
target_link_libraries(core-standard PRIVATE leanclr)
set_target_properties(core-standard PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
if(MSVC)
    target_compile_options(core-standard PRIVATE /EHs-c-)
else()
    target_compile_options(core-standard PRIVATE -fno-exceptions)
endif()
enable_testing()
if(EMSCRIPTEN)
    set_target_properties(core-standard PROPERTIES SUFFIX ".js")
    target_link_options(core-standard PRIVATE -sENVIRONMENT=node -sFILESYSTEM=0)
endif()
add_test(NAME core-standard COMMAND core-standard 10000)
