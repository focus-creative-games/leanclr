cmake_minimum_required(VERSION 3.15)
project(aot_hotspot_tests C CXX)
set(HOTSPOT_AOT_CPP "" CACHE PATH "Generated hotspot fixture sources")
set(HOTSPOT_DLLS "" CACHE PATH "Fixture and BCL directory")
add_subdirectory(../../runtime runtime)
file(GLOB generated CONFIGURE_DEPENDS "${HOTSPOT_AOT_CPP}/*.cpp")
add_executable(hotspot-aot main.cpp ${generated})
target_link_libraries(hotspot-aot PRIVATE leanclr)
target_compile_definitions(hotspot-aot PRIVATE HOTSPOT_AOT=1)
target_include_directories(hotspot-aot PRIVATE "${HOTSPOT_AOT_CPP}")
add_executable(hotspot-interp main.cpp)
target_link_libraries(hotspot-interp PRIVATE leanclr)
enable_testing()
add_test(NAME hotspot-aot COMMAND hotspot-aot "${HOTSPOT_DLLS}")
add_test(NAME hotspot-interpreter COMMAND hotspot-interp "${HOTSPOT_DLLS}")
