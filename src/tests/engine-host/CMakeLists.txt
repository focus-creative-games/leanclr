cmake_minimum_required(VERSION 3.15)
project(engine_host_contracts C CXX)
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "BCL adapter")
add_subdirectory(../../runtime runtime_build)
add_executable(engine-host main.cpp header.c)
target_include_directories(engine-host PRIVATE ../../host-bridge)
target_link_libraries(engine-host PRIVATE leanclr)
if(NOT EMSCRIPTEN)
    find_package(Threads REQUIRED)
    target_link_libraries(engine-host PRIVATE Threads::Threads)
endif()
if(WIN32 AND BUILD_SHARED_LEANCLR)
    add_custom_command(TARGET engine-host POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:leanclr> $<TARGET_FILE_DIR:engine-host>)
endif()
set(LEANCLR_ENGINE_RUNTIME_DIR "" CACHE STRING "CoreLib directory")
set(LEANCLR_ENGINE_MANAGED_DIR "" CACHE STRING "Managed engine contract directory")
enable_testing()
foreach(case lifecycle godot unreal invalid retry missing loader oom limit signature varargs generic initialize shutdown)
    add_test(NAME engine-${case} COMMAND engine-host ${case} "${LEANCLR_ENGINE_RUNTIME_DIR}" "${LEANCLR_ENGINE_MANAGED_DIR}")
    set_tests_properties(engine-${case} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
endforeach()
if(EMSCRIPTEN)
    set_target_properties(engine-host PROPERTIES SUFFIX ".js")
    target_link_options(engine-host PRIVATE -O2 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
endif()
