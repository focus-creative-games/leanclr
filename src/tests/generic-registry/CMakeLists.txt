cmake_minimum_required(VERSION 3.15)
project(global_generic_tests C CXX)
set(GENERIC_REGISTRY_CPP "" CACHE PATH "Generated sources")
set(GENERIC_REGISTRY_DLLS "" CACHE STRING "Managed assembly path in host/VFS")
option(GENERIC_REGISTRY_DISABLED "Expect no generic AOT instances" OFF)
option(GENERIC_REGISTRY_SHARED_REFERENCES "Expect canonical reference generic bodies" OFF)
add_subdirectory(../../runtime runtime)
file(GLOB generated CONFIGURE_DEPENDS "${GENERIC_REGISTRY_CPP}/*.cpp")
add_executable(generic-registry main.cpp registry_checks.cpp ${generated})
target_include_directories(generic-registry PRIVATE "${GENERIC_REGISTRY_CPP}")
target_link_libraries(generic-registry PRIVATE leanclr)
target_compile_features(generic-registry PRIVATE cxx_std_17)
if(GENERIC_REGISTRY_SHARED_REFERENCES)
    target_compile_definitions(generic-registry PRIVATE GENERIC_REGISTRY_SHARED_REFERENCES=1)
endif()
if(GENERIC_REGISTRY_DISABLED)
    target_compile_definitions(generic-registry PRIVATE GENERIC_REGISTRY_DISABLED=1)
endif()
if(EMSCRIPTEN)
    set_target_properties(generic-registry PROPERTIES SUFFIX ".js")
    target_link_options(generic-registry PRIVATE -O2 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
endif()
enable_testing()
if(GENERIC_REGISTRY_PROFILE_OUTPUT)
    add_test(NAME managed-aot-generics COMMAND generic-registry "${GENERIC_REGISTRY_DLLS}" profile "${GENERIC_REGISTRY_PROFILE_OUTPUT}")
else()
    add_test(NAME managed-aot-generics COMMAND generic-registry "${GENERIC_REGISTRY_DLLS}")
endif()
set_tests_properties(managed-aot-generics PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
