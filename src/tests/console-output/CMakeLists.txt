cmake_minimum_required(VERSION 3.15)
project(console_output_contracts C CXX)
add_subdirectory(../../tools/leanrun leanrun_build)
set(LEANCLR_CONSOLE_MANAGED_DIR "" CACHE PATH "Managed test assembly directory")
set(LEANCLR_CONSOLE_RUNTIME_DIR "" CACHE PATH "Matching BCL directory")
if(EMSCRIPTEN)
    foreach(target leanrun)
    set_target_properties(${target} PROPERTIES SUFFIX ".js")
    target_link_options(${target} PRIVATE -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=1 -lnodefs.js
        "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/aot-baseline/nodefs.js")
    endforeach()
endif()
enable_testing()
if(LEANCLR_BCL_ADAPTER STREQUAL "MONO" AND LEANCLR_HOST STREQUAL "PORTABLE")
    message(STATUS "Console contracts require a host providing Mono console initialization services")
    return()
endif()
foreach(entry Redirect Unicode Error Throw Restore Default)
    add_test(NAME reflection-${entry} COMMAND leanrun -l "${LEANCLR_CONSOLE_MANAGED_DIR}" -l "${LEANCLR_CONSOLE_RUNTIME_DIR}"
        -e "ConsoleOutputTests.Program::${entry}" ConsoleOutputTests)
    set_tests_properties(reflection-${entry} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
endforeach()

set_tests_properties(reflection-Default PROPERTIES PASS_REGULAR_EXPRESSION "console-default-string[\r\n]+123")
